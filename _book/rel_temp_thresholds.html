16  Temperature thresholds

This chapter will look at temperature thresholds and relationships to abundance indices.

Relationships will only be examined in years where we have data from all life stages (using the variable abundanceIndex_commonTimePeriod).

Code 
library(tidyverse)
library(DescTools)   # for SpearmanRho
library(corrplot)    # for corrplot
library(plotly)

source(here::here("functions.R"))

in_path <- here::here("data", "processed")

Read in data; trim to only common years; put life stages in proper order.

Code 
# brown
brn_all <- read.csv(here::here(in_path, "brn_abundAndEnv.csv"))
brn <- brn_all |> 
    filter(shrimp_year %in% common_shrimpYears) |> 
    mutate(life_stage = factor(life_stage, levels = ls_order_brn)) |> 
    rename(abundIndex = abundanceIndex_commonTimePeriod)
brn_wide <- brn |> 
    select(shrimp_year, life_stage, abundIndex) |> 
    pivot_wider(names_from = life_stage,
                values_from = abundIndex)

# white
wht_all <- read.csv(here::here(in_path, "wht_abundAndEnv.csv"))
wht <- wht_all |> 
    filter(shrimp_year %in% common_shrimpYears) |> 
    mutate(life_stage = factor(life_stage, levels = ls_order_wht)) |> 
    rename(abundIndex = abundanceIndex_commonTimePeriod)
wht_wide <- wht |> 
    select(shrimp_year, life_stage, abundIndex) |> 
    pivot_wider(names_from = life_stage,
                values_from = abundIndex)

16.1 Brown Shrimp

Code 
totalDays <- names(brn)[str_starts(names(brn), "totalDays")]
comp_grid <- expand.grid(life_stage = levels(brn$life_stage),
                         totalDays = totalDays)
comp_grid <- comp_grid |> 
    mutate(longestSpell = str_replace(totalDays, "totalDays", "longestSpell"))

comp_out <- list()

for(i in 1:nrow(comp_grid)){
    comp.stage <- comp_grid[i, "life_stage"]
    comp.totalDays <- comp_grid[i, "totalDays"]
    comp.longestSpell <- comp_grid[i, "longestSpell"]
    
    # subset the data frame
    comp.rows <- which(brn$life_stage == comp.stage)
    comp.cols <- which(names(brn) %in% c("shrimp_year", "abundIndex", as.character(comp.totalDays), as.character(comp.longestSpell)))
    
    
    comb <- brn[comp.rows, comp.cols]
    names(comb) <- c("shrimp_year", "abund_index", "totalDays", "longestSpell")
    
    # correlation and CI for total # days
    corr_totalDays <- SpearmanRho(comb$totalDays,
                                  comb$abund_index,
                                  use = "pairwise.complete.obs",
                                  conf.level = 0.95)
    corr_totalDays <- data.frame("totalDays_estimate" = corr_totalDays["rho"],
                                 "totalDays_ciLow" = corr_totalDays["lwr.ci"],
                                 "totalDays_ciHigh" = corr_totalDays["upr.ci"],
                                 row.names = NULL)
    
    # correlation and CI for longest # days
    corr_mostDays <- SpearmanRho(comb$longestSpell,
                                 comb$abund_index,
                                 use = "pairwise.complete.obs",
                                 conf.level = 0.95)
    corr_mostDays <- data.frame("consecDays_estimate" = corr_mostDays["rho"],
                                "consecDays_ciLow" = corr_mostDays["lwr.ci"],
                                "consecDays_ciHigh" = corr_mostDays["upr.ci"],
                                row.names = NULL)
    
    corr_out <- cbind(comp_grid[i, ],
                      corr_totalDays,
                      corr_mostDays)
    
    comp_out[[i]] <- corr_out
}

corrs_all <- bind_rows(comp_out) |> 
    mutate(totalDays_significant = case_when(totalDays_ciLow * totalDays_ciHigh > 0 ~ "yes",
                                            .default = "no"),
           consecDays_significant = case_when(consecDays_ciLow * consecDays_ciHigh > 0 ~ "yes",
                                            .default = "no"),
           thresholdC = as.numeric(str_remove(totalDays, "totalDays_lessThan")))
Code 
p <- ggplot(corrs_all,
       aes(x = thresholdC,
           y = totalDays_estimate,
           color = totalDays_significant)) +
    geom_linerange(aes(ymin = totalDays_ciLow,
                       ymax = totalDays_ciHigh),
                   col = "gray60") +
    geom_point(aes(size = totalDays_significant),
               alpha = 0.9) +
    scale_size_manual(values = c(2, 3),
                      guide = "none") +
    scale_color_manual(values = c("#a6611a", "#018571")) +   # r color brewer BrBG, colorblind-safe, endpoints of 5-class palette
    geom_hline(yintercept = 0) +
    facet_wrap(~ life_stage, ncol = 1) +
    theme_bw() +
    labs(title = "Brown Shrimp vs. Total Cold Days",
         x = "Temperature threshold (C)",
         y = "Correlation estimate and 95% CI (Spearman's rho)",
         col = "Significant?")

plotly_out(p, legend_pos = "right")
